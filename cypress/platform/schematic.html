Fixture B — modules and sides

  direction LR

  in  clk : clock
  in  a
  in  b
  out q

    out y
  end

  module Reg8
    in  clk : clock
    in  d
    out q
  end

  ALU  u_alu
  Reg8 u_q

  b --> u_alu.b
  u_alu.y --> u_q.d

    end
  end

  module Consumer
    group s_axis @left
      in  tvalid
      out tready
      in  tdata
    end
  end

  Consumer u_sink

  u_src.tvalid  --> u_sink.tvalid
  u_sink.tready --> u_src.tready
